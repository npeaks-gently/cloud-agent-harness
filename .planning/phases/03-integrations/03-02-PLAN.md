let cachedApiKey: string | undefined;

export async function getAnthropicApiKey(): Promise<string> {
  if (cachedApiKey) return cachedApiKey;
  const secretArn = process.env.ANTHROPIC_API_KEY_SECRET_ARN;
  if (!secretArn) throw new Error('ANTHROPIC_API_KEY_SECRET_ARN not set');
  const smClient = new SecretsManagerClient({
    region: process.env.AWS_DEFAULT_REGION ?? 'us-east-1',
  });
  const response = await smClient.send(
    new GetSecretValueCommand({ SecretId: secretArn }),
  );
  if (!response.SecretString) {
    throw new Error('Secrets Manager returned empty SecretString');
  }
  cachedApiKey = response.SecretString;
  return cachedApiKey;
}
```

// --- Error ---
LinearClientError extends Error { constructor(message, operation, issueId?) }

// --- Secrets ---
let cachedLinearApiKey: string | undefined;
export async function getLinearApiKey(): Promise<string>
  - Reads LINEAR_API_KEY_SECRET_ARN env var
  - Same Secrets Manager cache pattern

// --- Configuration ---
// Linear team ID and state IDs are deployment-specific (Claude's Discretion).
// Read from environment variables on the Lambda.
function getLinearConfig(): { teamId: string; states: Record<string, string> }
  - Returns:
    teamId: process.env.LINEAR_TEAM_ID ?? ''
    states: JSON.parse(process.env.LINEAR_STATE_MAP ?? '{}')
      Expected shape: { "todo": "state-uuid", "in_progress": "state-uuid", "done": "state-uuid" }

// --- Ticket CRUD ---
export async function createParentTicket(
  runId: string,
  featureDescription: string,
): Promise<{ ticketId: string; identifier: string }>
  - Creates LinearClient with getLinearApiKey()
  - Calls linearClient.createIssue({
      teamId: getLinearConfig().teamId,
      title: `[CAH] ${featureDescription}`,
      description: `Pipeline run: ${runId}\n\nFeature: ${featureDescription}`,
    })
  - Returns { ticketId: issue.id, identifier: issue.identifier } from the IssuePayload.issue
  - try/catch wraps SDK call, throws LinearClientError
  - CloudWatch log: level info, message 'Linear parent ticket created'

export async function createSubTicket(
  parentTicketId: string,
  phaseNumber: number,
  phaseName: string,
): Promise<{ ticketId: string; identifier: string }>
  - Calls linearClient.createIssue({
      teamId: getLinearConfig().teamId,
      title: `Phase ${phaseNumber}: ${phaseName}`,
      parentId: parentTicketId,
    })
  - Returns { ticketId, identifier }
  - Per D-10: sub-ticket linked via parentId

export async function updateTicketStatus(
  ticketId: string,
  statusKey: string,
): Promise<void>
  - statusKey is a key into LINEAR_STATE_MAP (e.g., 'in_progress', 'done')
  - Calls linearClient.updateIssue(ticketId, { stateId: getLinearConfig().states[statusKey] })
  - Per D-10: stage transitions update active phase's sub-ticket status

export async function attachPrUrl(
  ticketId: string,
  prUrl: string,
): Promise<void>
  - Calls linearClient.createAttachment({
      issueId: ticketId,
      title: 'Pull Request',
      url: prUrl,
    })
  - Per D-11: PR URL linked to parent ticket on completion
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Lambda -> Slack API | Bot token authenticates; messages sent to configured channel |
| Lambda -> GitHub API | PAT authenticates; scoped to target repo contents + PRs |
| Lambda -> Linear API | API key authenticates; full issue CRUD access |
| Lambda -> Secrets Manager | IAM role-based access; secrets never logged |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-05 | Information Disclosure | slack.ts, github.ts, linear.ts | mitigate | API keys fetched from Secrets Manager at cold start; never stored in Lambda env vars as plaintext; never logged (error messages include operation name, not credentials) |
| T-03-06 | Spoofing | github.ts createFeatureBranch | mitigate | GitHub token scoped to specific repo via fine-grained PAT; branch creation requires Contents write permission |
| T-03-07 | Tampering | linear.ts updateTicketStatus | accept | Linear API key grants full issue access; acceptable for v1 solo team; service account recommended for production |
| T-03-08 | Denial of Service | All integration utilities | accept | External API failures caught and re-thrown as typed errors; pipeline fails gracefully; retries handled by SQS redelivery |
</threat_model>
